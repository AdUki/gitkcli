# tuikit demo 2 (demos/demo_2_widgets.py): segmented rows (on/off switch,
# choice, a radio group of toggles), a keyboard-navigable button row, and the
# input / confirm / error dialogs.
size      90x18
demo      demo_2_widgets
capture   start
# Enter flips the switch; Down + Enter cycles the theme; on the Size row
# Right steps the radio group (M -> L -> XL) and Tab wraps around to <S>.
key       <Enter> <Down> <Enter> <Down> <Right> <Right> <Tab>
wait      stable
capture   size_keys
# Left on the switch row picks ON again; a click picks <L> (columns 79-81).
key       <Up> <Up> <Left>
mouse     click 80 4
wait      stable
capture   toggled
# G selects the button row; Right moves to [Rename]; Enter opens the prompt.
key       G <Right> <Enter>
wait      stable
key       <bspace>*5
text      "alice"
wait      stable
capture   rename_prompt
key       <Enter>
wait      stable
capture   renamed
# [Reset] asks first (focus on Cancel); y confirms and restores the defaults.
key       <Right> <Enter>
wait      stable
capture   confirm
key       y
wait      stable
capture   reset
wait      2.5
# [Fail] raises the red error dialog; Enter dismisses it.
key       <Right> <Enter>
wait      stable
capture   error
key       <Enter>
wait      stable
capture   dismissed
