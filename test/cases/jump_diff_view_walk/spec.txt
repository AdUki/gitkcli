# The jump list crosses panes. Open three commit diffs (each Enter records the
# log row it was opened from), then scroll inside the last one. Ctrl-Left leaves
# the diff pane for the log row that diff was opened from, and again for the
# previous one; Ctrl-Right walks forward, the last step landing back in the diff
# pane at the line it was left at (forward_to_diff == diff_third).
size      120x40
launch
key       <Enter>
wait      stable
key       q
key       <Down>*5
key       <Enter>
wait      stable
key       q
key       <Down>*5
key       <Enter>
wait      stable
key       j*8
wait      stable
capture   diff_third
key       C-Left
wait      stable
capture   back_to_log
key       C-Left
wait      stable
capture   back_further
key       C-Right
wait      stable
key       C-Right
wait      stable
capture   forward_to_diff
