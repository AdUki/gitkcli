# Launched outside a git repository (the fixture repo with its .git removed):
# instead of every git job failing, a dialog offers `git init` (the path shown
# home-relative). Init runs it and loads the now empty repository: its files are
# all untracked, so the "working directory" pseudo-row appears. Opening it lists
# them (and their content as new files); A stages them all, c opens the commit
# dialog, and Enter in the summary makes the first commit,
# which the log then shows with its branch.
size      120x40
config    default
run       rm -rf .git
launch
wait      stable
capture   init_dialog
key       <Enter>
wait      stable
capture   initialized
key       <Enter>
wait      stable
capture   untracked
key       A
wait      stable
capture   all_staged
key       c
wait      stable
text      "First commit"
capture   commit_dialog
key       <Enter>
wait      stable
key       q
wait      stable
capture   first_commit
