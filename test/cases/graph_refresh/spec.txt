# F5 refresh in --graph mode. The graph engine (gitk.graph) files the new
# commit into the loaded graph incrementally (`old..HEAD`, no full reload).
# Launch with --graph, create an external commit on
# HEAD, press F5, then 'g' to scroll to the (reloaded) top where the new commit
# is visible with correct graph art.
size      120x40
config    default
launch    --graph
wait      stable
run       git commit --allow-empty -m "external graph commit"
key       <F5>
wait      2.5
key       g
wait      stable
capture   reloaded
